1. EXECUTIVE SUMMARY & PARADIGM SHIFT

1.1 The Failure of Probabilistic Scanning Architecture

Standard low-cost hazard scouting prototypes (including legacy systems relying on simple Ultrasonic and MQ-series analog gas sensors) suffer from a fundamental flaw: Acoustic and Chemical Ambiguity. Ultrasonic sensors emit wide acoustic cones that bounce unpredictably off diagonal walls, while analog gas sensors suffer from heavy cross-sensitivity to humidity and alcohol. This results in software that outputs only a statistical guess of obstacle or hazard presence.

For tactical first responders and EOD personnel, a statistical guess is operationally useless. Commercial defense-grade LiDAR robots deliver high fidelity but cost upwards of ₹50 Lakhs to ₹80 Lakhs.

1.2 The A.R.G.U.S Solution

Project A.R.G.U.S replaces statistical estimation with a Frugal Swept Solid-State LiDAR Engine and Hardware Coincidence Matrix. The UGV uses four VL53L1X Time-of-Flight (ToF) laser sensors attached to four MG90S metal-gear servos across the chassis quadrants. When stopped, A.R.G.U.S executes a synchronous $360^\circ$ swept laser scan in 1.1 seconds, constructing a distortion-free 2D occupancy grid.

Simultaneously, hazard verification enforces a strict logical "AND" across independent physical domains: digital MEMS gas spectroscopy (SGP41 VOC/$NO_x$, DFRobot $NH_3$), inductive metallic shrapnel detection (LJ12A3), and infrared thermal mapping (MLX90614).

4. STEP-BY-STEP HARDWARE ASSEMBLY GUIDE

Phase 1: UGV Power Distribution & Drivetrain

  • Step 1.1: Connect the Pro-Range 11.1V 3S LiPo through the 6A SPST Rocker Switch. Split the main output into two lines: Raw 11.1V and Buck Input.
  • Step 1.2: Connect the raw 11.1V line to the L298N Motor Driver `VMS` terminal and the Brown wire of the **LJ12A3 Proximity Switch**.
  • Step 1.3: Wire the 11.1V line into the **LM2596 Buck Converter**. Adjust the potentiometer until `OUT+` reads exactly 5.0V DC. Route this 5.0V rail to the ESP32-S3 `VIN` pin and the VCC pins of all 4 MG90S Servos.
  • Step 1.4: Solder all ground wires (LiPo (-), LM2596 `OUT-`, L298N `GND`, ESP32 `GND`, and sensor GNDs) to a single Common Ground Bus on a perfboard.

Phase 2: Swept ToF Array & Sensor Installation

  • Step 2.1: Mount one MG90S servo at each chassis corner. Secure a VL53L1X ToF breakout to each servo horn.
  • Step 2.2: Wire all four ToF `SDA` and `SCL` pins in parallel to ESP32-S3 GPIO 8 and GPIO 9. Wire the four `XSHUT` control lines to GPIOs 33, 34, 35, and 36.
  • Step 2.3: Wire the 7Semi SGP41 to the shared I2C bus (GPIO 8/9). Wire the DFRobot NH3 Analog Output to ESP32-S3 GPIO 4. Connect the LJ12A3 Black signal wire to GPIO 6.

Phase 3: Handheld Remote Controller Assembly

  • Step 3.1: Mount the ESP32 38-Pin NodeMCU-32 onto a prototype board.
  • Step 3.2: Connect PS2 Joystick 1 (Drive Control: VRx $\rightarrow$ GPIO 34, VRy $\rightarrow$ GPIO 35) and PS2 Joystick 2 (Pan/Tilt: VRx $\rightarrow$ GPIO 32, VRy $\rightarrow$ GPIO 33).
  • Step 3.3: Wire the three Red PBS-11B Push Buttons to GPIOs 25, 26, and 27 with internal pull-ups enabled (`INPUT_PULLUP`). Assign Button 1 as the primary **360° SCAN** trigger.

5. APPLIED MATHEMATICS: COORDINATE TRANSFORMATION

Each swept sensor assembly acquisitions data in Polar Coordinates $(r, \theta)$. The sensor returns a direct distance measurement ($r$) to the obstacle at a specific global sweep angle ($\theta_{\text{global}}$):

$$\theta_{\text{global}} = \theta_{\text{robot\_heading}} + \theta_{\text{quadrant\_offset}} + \theta_{\text{servo\_sweep}}$$

To build a flat obstacle map used for point-to-point navigation, the polar data points $(r, \theta)$ are transformed into Cartesian Coordinates $(x, y)$ relative to the robot's static center point $(0,0)$:

$$x = r \times \cos(\theta_{\text{global}})$$ $$y = r \times \sin(\theta_{\text{global}})$$

/*

 * Architecture: ESP32-S3 Dual-Core (FreeRTOS)
 * Payload: 4x VL53L1X LiDAR, LJ12A3, SGP41, DFRobot NH3, CC1101 Radio

#include <Arduino.h>
#include <WiFi.h>
#include <esp_now.h>
#include <Wire.h>
#include <ESP32Servo.h>
#include <SparkFun_VL53L1X.h> 

#define PIN_I2C_SDA 8
#define PIN_I2C_SCL 9

#define XSHUT_FL 33 
#define XSHUT_FR 34 
#define XSHUT_RR 35 
#define XSHUT_RL 36 

#define PIN_SRV_FL 15
#define PIN_SRV_FR 16
#define PIN_SRV_RR 17
#define PIN_SRV_RL 18

#define PIN_NH3_ADC 4
#define PIN_METAL_SENS 6
#define PIN_LED_RED 2
#define PIN_LED_GREEN 3

Servo servoFL, servoFR, servoRR, servoRL;
SFEVL53L1X tofFL(Wire, XSHUT_FL);
SFEVL53L1X tofFR(Wire, XSHUT_FR);
SFEVL53L1X tofRR(Wire, XSHUT_RR);
SFEVL53L1X tofRL(Wire, XSHUT_RL);

TaskHandle_t LidarScanTask;
uint8_t occupancyGrid[100][100]; 

typedef struct __attribute__((packed)) {
  uint8_t threat_stage;    
  float nh3_ppm;           
  bool metal_present;      
} TelemetryPacket;

TelemetryPacket txData;
uint8_t remoteAddress[] = {0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF};

// --- I2C TOF ADDRESS REASSIGNMENT ---
void initToFSensors() {
  pinMode(XSHUT_FL, OUTPUT); pinMode(XSHUT_FR, OUTPUT);
  pinMode(XSHUT_RR, OUTPUT); pinMode(XSHUT_RL, OUTPUT);

  digitalWrite(XSHUT_FL, LOW); digitalWrite(XSHUT_FR, LOW);
  digitalWrite(XSHUT_RR, LOW); digitalWrite(XSHUT_RL, LOW);
  delay(10);

  digitalWrite(XSHUT_FL, HIGH); delay(10);
  tofFL.begin(); tofFL.setI2CAddress(0x30); 

  digitalWrite(XSHUT_FR, HIGH); delay(10);
  tofFR.begin(); tofFR.setI2CAddress(0x31);

  digitalWrite(XSHUT_RR, HIGH); delay(10);
  tofRR.begin(); tofRR.setI2CAddress(0x32);

  digitalWrite(XSHUT_RL, HIGH); delay(10);
  tofRL.begin(); tofRL.setI2CAddress(0x33);
}

// --- CARTESIAN MATH ROUTINE ---
void mapPointToGrid(int distance_mm, float global_angle_deg) {
  if(distance_mm == 0 || distance_mm > 4000) return; 

  float r_cells = distance_mm / 100.0;
  float rad = global_angle_deg * (PI / 180.0);

  int x = 50 + (int)(r_cells * cos(rad));
  int y = 50 + (int)(r_cells * sin(rad));

  if(x >= 0 && x < 100 && y >= 0 && y < 100) {
    occupancyGrid[x][y] = 1; 
  }
}

// --- CORE 1: SYNCHRONOUS SWEEP & MAP ---
void ScanTaskCode(void * pvParameters) {
  for (;;) {
    ulTaskNotifyTake(pdTRUE, portMAX_DELAY); 
    memset(occupancyGrid, 0, sizeof(occupancyGrid));

    const int SERVO_STEP = 2; 
    const int SETTLE_MS = 25; 

    for (int angle = 0; angle <= 90; angle += SERVO_STEP) {
      servoFL.write(angle); servoFR.write(angle);
      servoRR.write(angle); servoRL.write(angle);
      
      vTaskDelay(pdMS_TO_TICKS(SETTLE_MS));

      tofFL.startRanging(); tofFR.startRanging();
      tofRR.startRanging(); tofRL.startRanging();
      
      while(!tofFL.checkForDataReady() || !tofFR.checkForDataReady());
      
      int d1 = tofFL.getDistance(); int d2 = tofFR.getDistance();
      int d3 = tofRR.getDistance(); int d4 = tofRL.getDistance();
      
      tofFL.clearInterrupt(); tofFR.clearInterrupt();
      tofRR.clearInterrupt(); tofRL.clearInterrupt();

      mapPointToGrid(d1, angle + 45);        
      mapPointToGrid(d2, angle + 135);       
      mapPointToGrid(d3, angle + 225);       
      mapPointToGrid(d4, angle + 315);       
    }
    
    servoFL.write(0); servoFR.write(0);
    servoRR.write(0); servoRL.write(0);
    Serial.println("A.R.G.U.S SCAN COMPLETE");
  }
}

// --- CORE 0: NETWORK & COINCIDENCE MATRIX ---
void setup() {
  Serial.begin(115200);
  Wire.begin(PIN_I2C_SDA, PIN_I2C_SCL);

  pinMode(PIN_METAL_SENS, INPUT_PULLUP);
  pinMode(PIN_LED_RED, OUTPUT);
  pinMode(PIN_LED_GREEN, OUTPUT);

  initToFSensors();
  
  servoFL.attach(PIN_SRV_FL); servoFR.attach(PIN_SRV_FR);
  servoRR.attach(PIN_SRV_RR); servoRL.attach(PIN_SRV_RL);

  xTaskCreatePinnedToCore(ScanTaskCode, "LidarScanTask", 10000, NULL, 1, &LidarScanTask, 1);

  WiFi.mode(WIFI_STA);
  esp_now_init();
}

void loop() {
  float rawNH3 = analogRead(PIN_NH3_ADC);
  txData.nh3_ppm = (rawNH3 / 4095.0) * 300.0;
  bool chemAnomaly = (txData.nh3_ppm > 50.0);
  txData.metal_present = (digitalRead(PIN_METAL_SENS) == LOW);

  if (chemAnomaly && txData.metal_present) {
    txData.threat_stage = 3; 
    digitalWrite(PIN_LED_RED, HIGH); digitalWrite(PIN_LED_GREEN, LOW);
  } else {
    txData.threat_stage = 0; 
    digitalWrite(PIN_LED_RED, LOW); digitalWrite(PIN_LED_GREEN, HIGH);
  }

  esp_now_send(remoteAddress, (uint8_t *) &txData, sizeof(txData));
  delay(100); 
}
